def to_z(v):
    u=(1-Z)/(1+Z)
    def pol(p):
        parity={m[1]%2 for m in p}
        if len(parity)>1:raise ValueError('Mixed parity')
        b=next(iter(parity),0)
        return sum((c*R**m[0]*u**((m[1]-b)//2) for m,c in p.items()),a.F.zero),b
    d={};odd={}
    for ph,rat in v.d.items():
        n,p=pol(rat.numer);de,b=pol(rat.denom)
        if p==b:d[ph]=n/de
        elif p-b==1:odd[ph]=n/de/(1+Z)
        else:odd[ph]=n/de/(1-Z)
    return a.Coeff(d),a.Coeff(odd)

def expr(v):
    z=sp.Symbol('z',real=True)
    return (v[0].expr()+sp.sqrt(1-a.HALF_ANGLE**2)*v[1].expr()).subs(a.HALF_ANGLE,z)
def tex(x):return sp.latex(x,fold_short_frac=False,order='lex')
def equation(body,label):return '\\begin{equation}\n'+body+'\n\\label{'+label+'}\n\\end{equation}\n'

def multiline_sum(ex,width=65):
    terms=sp.Add.make_args(sp.expand(ex));lines=[];current=''
    for term in terms:
        negative=term.could_extract_minus_sign();s=tex(-term if negative else term)
        addition=(' - ' if negative else (' + ' if current or lines else ''))+s
        # approximate the number of printed glyphs, rather than LaTeX bytes
        measure=re.sub(r'\\[A-Za-z]+','X',current+addition)
        if current and len(measure)>width:lines.append(current);current=addition
        else:current+=addition
    if current:lines.append(current)
    if len(lines)==1:return lines[0]
    return '\\shortstack[l]{'+'\\\\\n'.join('$\\displaystyle '+line+'$' for line in lines)+'}'

def rational_equation(ex,number):
    # Coefficients have a single phase in almost all cases. A common denominator
    # gives a stable, fully explicit numerator without new LaTeX macros.
    n,d=sp.fraction(sp.factor(ex));n=sp.expand(n)
    if len(tex(ex))<185:return equation(f'R_{{{number}}}(r,z)='+tex(ex),f'eq:cat-rational-{number}')
    body=f'R_{{{number}}}(r,z)=\\frac{{N_{{{number}}}(r,z)}}{{D_{{{number}}}(r,z)}}.'
    out=equation(body,f'eq:cat-rational-{number}')
    out+=equation(f'N_{{{number}}}(r,z)='+multiline_sum(n,62),f'eq:cat-numerator-{number}')
    out+=equation(f'D_{{{number}}}(r,z)='+tex(sp.factor(d)),f'eq:cat-denominator-{number}')
    return out

\section{Область применения и индексация}
Каталог относится к радиационной ветви $U=0$ и к кинематике независимых радиусов $r=k'/k>0$, $-1<z<1$, $x=k\eta$. Это не общая гравитационная амплитуда на произвольном фоне. В формулах статьи сохраняются внешние частотные множители, общий масштаб и отдельная объёмная нормировка строго однородных линий.

Индексы позиций $0,1,2,3$ соответствуют позициям $1,2,3,4$ статьи. Индексы полей $0,1,2$ означают скаляр, первую и вторую линейные тензорные поляризации. Направленные внутрь импульсы в единицах $k$ равны
\begin{equation}
\shortstack[l]{$\displaystyle \mathbf p_0=(0,0,1),\quad \mathbf p_1=(0,0,-1),$\\
$\displaystyle \mathbf p_2=(-r\sqrt{1-z^2},0,-rz),\quad \mathbf p_3=(r\sqrt{1-z^2},0,rz).$}
\label{eq:cat-momenta}
\end{equation}
Знаковые частоты равны $(c_{A_0},c_{A_1},-rc_{A_2},-rc_{A_3})$, где $c_0=1/\sqrt3$, $c_1=c_2=1$. Для каждого внешнего импульса $\mathbf n=\mathbf p/|\mathbf p|$ используются $\mathbf e=(0,1,0)$, $\mathbf b=(-n_z,0,n_x)$, $E_1=\mathbf e\mathbf e-\mathbf b\mathbf b$, $E_2=\mathbf e\mathbf b+\mathbf b\mathbf e$; квадраты обеих норм равны двум. Параметр исходного алгебраического вычисления $t_\theta$ связан с $z$ равенством $t_\theta^2=(1-z)/(1+z)$.

Для внутреннего импульса $\mathbf p$ ненулевой нормы используется ненормированный $\mathbf b=(-p_z,0,p_x)$: $E_1=\mathbf e\mathbf e-\mathbf b\mathbf b/p^2$, $E_2=\mathbf e\mathbf b+\mathbf b\mathbf e$. Квадраты норм скаляра и двух тензоров равны $1,2,2p^2$. Переход от $\mathbf p$ к $-\mathbf p$ между концами даёт множители $\tau_I=(1,1,-1)$.

При $\mathbf p=0$ используются пять симметричных бесследовых матриц. Две диагональные матрицы имеют вид $\mathrm{diag}(1,-1,0)$ и $\mathrm{diag}(1,1,-2)$. Остальные три имеют единицы в парах позиций $xy,yx$, $xz,zx$ и $yz,zy$ соответственно. Квадраты их норм равны $2,6,2,2,2$. Кинетический знак однородной скалярной координаты равен $-1$. Ковариации геометрических координат делятся на положительный квадрат геометрической нормы; скалярный знак учитывается свободным движением, а не отрицательной вероятностью состояния.

\begin{equation}
\frac{H_4}{g_{\mathrm R}^2k^4}=\sum_{p=-4}^{0}h_px^p,\qquad h_p=b_p+u_p+v_p.
\label{eq:cat-contact-parts}
\end{equation}
Коэффициенты при неуказанных степенях равны нулю. Обозначения $R_s$ в следующих формулах означают явные алгебраические функции раздела~\ref{sec:cat-rationals}, а не внешние электронные записи. При записи через $z$ некоторые источники глобальных декартовых тензоров содержат $\sqrt{1-z^2}$; после перехода к $t_\theta$ все функции рациональны.

Пусть $T$ переставляет входящие и исходящие пары, а $S$ переставляет две позиции внутри одной пары. Точные правила имеют вид
\begin{equation}
 h_p^{B A C D}(r,z)=h_p^{A B D C}(r,z)=h_p^{A B C D}(r,-z),\qquad
 h_p^{C D A B}(r,z)=r^{4+p}\overline{h_p^{A B C D}(1/r,z)}.
\label{eq:cat-contact-symmetries}
\end{equation}
Сначала упорядочиваются индексы внутри каждой пары, затем пары упорядочиваются лексикографически. Нечётное число внутренних перестановок даёт $z\mapsto-z$. Перестановка пар даёт $r\mapsto1/r$, комплексное сопряжение и множитель $r^{4+p}$. Во всех 81 компонентах эти тождества проверены отдельно.
