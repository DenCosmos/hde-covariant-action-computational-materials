#!/usr/bin/env python3
"""Воспроизвести редакционно проверенную V40 по окончательным 22 заменам.

По умолчанию читает source/article_v39.tex и записывает article_v40.tex.
Математические коэффициенты и PDF не пересчитывает. Манифест не изменяет.
"""
from __future__ import annotations
import argparse
import hashlib
import json
from pathlib import Path
import sys

BASE = Path(__file__).resolve().parents[1]


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--source', type=Path, default=BASE / 'source/article_v39.tex')
    parser.add_argument('--output', type=Path, default=BASE / 'article_v40.tex')
    args = parser.parse_args()
    try:
        manifest = json.loads((BASE / 'patches/replacements_v39_to_v40.json').read_text(encoding='utf-8'))
        source = args.source.read_bytes()
        if hashlib.sha256(source).hexdigest() != manifest['base_sha256']:
            raise ValueError('Исходник отличается от проверенной V39; файл не изменён.')
        text = source.decode('utf-8')
        intervals: list[tuple[int, int, str]] = []
        for patch in manifest['patches']:
            if text.count(patch['find']) != 1:
                raise ValueError(f"Неоднозначный поиск для {patch['id']}.")
            start = text.index(patch['find'])
            intervals.append((start, start + len(patch['find']), patch['replace']))
        intervals.sort()
        if any(left[1] > right[0] for left, right in zip(intervals, intervals[1:])):
            raise ValueError('Фрагменты поиска пересекаются.')
        for start, end, replacement in reversed(intervals):
            text = text[:start] + replacement + text[end:]
        result = text.encode('utf-8')
        if hashlib.sha256(result).hexdigest() != manifest['result_sha256']:
            raise ValueError('Контрольная сумма результата не совпадает; файл не записан.')
        args.output.parent.mkdir(parents=True, exist_ok=True)
        args.output.write_bytes(result)
        print(f'Создан {args.output}: {len(intervals)} замен, SHA-256 проверена.')
        return 0
    except (OSError, ValueError, KeyError) as exc:
        print(f'Ошибка: {exc}', file=sys.stderr)
        return 1


if __name__ == '__main__':
    raise SystemExit(main())
